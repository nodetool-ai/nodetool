cmake_minimum_required(VERSION 3.20)
project(nodetool_workflow LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
set(NW_BUILD_ID "")
foreach(NW_SOURCE CMakeLists.txt compiler.cpp compiler.hpp gpu.cpp gpu.hpp runtime.cpp runtime.hpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${NW_SOURCE}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${NW_SOURCE}" NW_SOURCE_HASH)
  string(APPEND NW_BUILD_ID "${NW_SOURCE_HASH}")
endforeach()
string(SHA256 NW_BUILD_HASH "${NW_BUILD_ID}")
add_executable(nodetool-workflow main.cpp runtime.cpp compiler.cpp gpu.cpp)
target_compile_features(nodetool-workflow PRIVATE cxx_std_17)
target_compile_definitions(nodetool-workflow PRIVATE NW_COMPILER_BUILD_HASH="${NW_BUILD_HASH}")
target_include_directories(nodetool-workflow SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(nodetool-workflow PRIVATE ${LLVM_DEFINITIONS})
if(TARGET LLVM)
  set(NW_LLVM_LIBS LLVM)
else()
  llvm_map_components_to_libnames(NW_LLVM_LIBS Support TargetParser)
endif()
target_link_libraries(nodetool-workflow PRIVATE ${NW_LLVM_LIBS} ${CMAKE_DL_LIBS})
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(workflow-runtime-test tests/runtime_test.cpp runtime.cpp gpu.cpp)
  target_compile_features(workflow-runtime-test PRIVATE cxx_std_17)
  target_include_directories(workflow-runtime-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_include_directories(workflow-runtime-test SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
  target_compile_definitions(workflow-runtime-test PRIVATE ${LLVM_DEFINITIONS})
  target_link_libraries(workflow-runtime-test PRIVATE ${NW_LLVM_LIBS} ${CMAKE_DL_LIBS} Threads::Threads)
  add_test(NAME workflow-runtime COMMAND workflow-runtime-test $<TARGET_FILE:nodetool-workflow>)
  add_test(NAME workflow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_workflow.py $<TARGET_FILE:nodetool-workflow>)
endif()
