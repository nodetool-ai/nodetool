cmake_minimum_required(VERSION 3.20)
project(nodetool_workflow LANGUAGES C CXX)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.__path__[0]); print(torch.__version__); print(int(torch.compiled_with_cxx11_abi()))"
  OUTPUT_VARIABLE NW_TORCH_INFO OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\n" ";" NW_TORCH_INFO "${NW_TORCH_INFO}")
list(GET NW_TORCH_INFO 0 NW_TORCH_ROOT)
list(GET NW_TORCH_INFO 1 NW_TORCH_VERSION)
list(GET NW_TORCH_INFO 2 NW_TORCH_ABI)
find_package(LLVM REQUIRED CONFIG)
add_executable(nodetool-workflow main.cpp runtime.cpp)
target_compile_features(nodetool-workflow PRIVATE cxx_std_17)
target_compile_definitions(nodetool-workflow PRIVATE NW_TORCH_VERSION="${NW_TORCH_VERSION}")
target_include_directories(nodetool-workflow SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_include_directories(nodetool-workflow SYSTEM PRIVATE "${NW_TORCH_ROOT}/include" "${NW_TORCH_ROOT}/include/torch/csrc/api/include")
target_compile_definitions(nodetool-workflow PRIVATE _GLIBCXX_USE_CXX11_ABI=${NW_TORCH_ABI})
target_link_directories(nodetool-workflow PRIVATE "${NW_TORCH_ROOT}/lib")
set_target_properties(nodetool-workflow PROPERTIES BUILD_RPATH "${NW_TORCH_ROOT}/lib")
if(TARGET LLVM)
  set(NW_LLVM_LIBS LLVM)
else()
  llvm_map_components_to_libnames(NW_LLVM_LIBS Support)
endif()
target_link_libraries(nodetool-workflow PRIVATE torch torch_cpu c10 ${NW_LLVM_LIBS} ${CMAKE_DL_LIBS})
