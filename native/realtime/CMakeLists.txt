cmake_minimum_required(VERSION 3.24)
project(NodeToolRealtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
include(FetchContent)
if(NOT TARGET nlohmann_json::nlohmann_json)
  FetchContent_Declare(nlohmann_json
    URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
    URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
  FetchContent_MakeAvailable(nlohmann_json)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS schema/plan-v1.schema.json)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/schema/plan-v1.schema.json NT_PLAN_SCHEMA)
# MSVC limits each literal token. Adjacent raw literals preserve the same JSON.
string(REPLACE "\n" ")nt_schema\"\nR\"nt_schema(" NT_PLAN_SCHEMA "${NT_PLAN_SCHEMA}")
configure_file(src/plan_schema.h.in generated/plan_schema.h @ONLY)
add_library(nodetool_realtime STATIC src/plan.cpp src/control.cpp src/math.cpp)
target_include_directories(nodetool_realtime PUBLIC include PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(nodetool_realtime PUBLIC nlohmann_json::nlohmann_json)
add_executable(realtime-plan-test tests/plan_test.cpp)
target_link_libraries(realtime-plan-test PRIVATE nodetool_realtime)
target_compile_definitions(realtime-plan-test PRIVATE NT_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME realtime-plan-loader COMMAND realtime-plan-test)
set_tests_properties(realtime-plan-loader PROPERTIES TIMEOUT 60)
add_executable(realtime-control-test tests/control_test.cpp)
target_link_libraries(realtime-control-test PRIVATE nodetool_realtime)
target_include_directories(realtime-control-test PRIVATE ../timeline/src)
add_test(NAME realtime-control COMMAND realtime-control-test)
set_tests_properties(realtime-control PROPERTIES TIMEOUT 60)
foreach(target nodetool_realtime realtime-plan-test realtime-control-test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
  endif()
endforeach()

  else()

  endif()
