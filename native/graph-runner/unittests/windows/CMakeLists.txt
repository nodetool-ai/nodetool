cmake_minimum_required(VERSION 3.28)
project(GraphRunnerWindowsFiles LANGUAGES CXX)
enable_testing()
add_executable(gr_windows_files platform_fs_test.cpp ../../src/host/platform_fs_windows.cpp ../../src/host/paths.cpp)
target_compile_features(gr_windows_files PRIVATE cxx_std_20)
target_compile_definitions(gr_windows_files PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(gr_windows_files PRIVATE ../../src)
target_link_libraries(gr_windows_files PRIVATE ntdll bcrypt)
add_test(NAME gr_windows_files COMMAND gr_windows_files)
find_program(NODE_EXECUTABLE node REQUIRED)
add_test(NAME gr_windows_junction COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/junction-test.mjs" "$<TARGET_FILE:gr_windows_files>")
