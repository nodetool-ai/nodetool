cmake_minimum_required(VERSION 3.22)
project(NodeToolGameBrowser VERSION 0.1.0 LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Use the pinned Emscripten SDK to configure this target")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GAME_BUILD_DEPENDENCY_PROBES OFF CACHE BOOL "" FORCE)
set(GAME_SANITIZERS OFF)
add_compile_options(-fexceptions)
add_link_options(-fexceptions)
set(GAME_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(json)
include("${GAME_SOURCE_DIR}/cmake/Portable.cmake")
get_target_property(GAME_JOLT_SOURCES Jolt SOURCES)
list(FILTER GAME_JOLT_SOURCES EXCLUDE REGEX "/Core/JobSystemThreadPool\\.(cpp|h)$")
set_property(TARGET Jolt PROPERTY SOURCES ${GAME_JOLT_SOURCES})
add_executable(native-game "${GAME_SOURCE_DIR}/src/browser/Bridge.cpp" "${GAME_SOURCE_DIR}/src/browser/Model.cpp")
target_link_libraries(native-game PRIVATE game_service)
set_target_properties(native-game PROPERTIES SUFFIX ".mjs")
target_link_options(native-game PRIVATE
  --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createNativeGameModule
  -sENVIRONMENT=web,worker,node -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=67108864
  -sMAXIMUM_MEMORY=536870912 -sSTACK_SIZE=2097152 -sABORTING_MALLOC=0
  "-sEXPORTED_FUNCTIONS=['_ng_create','_ng_dispatch','_ng_response_pointer','_ng_response_length','_ng_destroy','_malloc','_free']"
  -sFORCE_FILESYSTEM=1 "-sEXPORTED_RUNTIME_METHODS=['FS','HEAPU8']")
