# Deliberate defects. Never part of a normal build or normal test run.
function(nt_plant name)
    add_executable(nt_plant_${name} ${name}.cpp)
    set_target_properties(nt_plant_${name} PROPERTIES AUTOMOC OFF)
    add_test(NAME nt_plant_${name} COMMAND nt_plant_${name})
endfunction()
if(NT_FUZZ)
    add_executable(nt_plant_fuzz fuzz.cpp)
    set_target_properties(nt_plant_fuzz PROPERTIES AUTOMOC OFF)
    target_link_options(nt_plant_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
elseif(NT_SANITIZE STREQUAL "address,undefined")
    nt_plant(uaf)
    nt_plant(ubsan)
    nt_plant(leak)
elseif(NT_SANITIZE STREQUAL "thread")
    find_package(Threads REQUIRED)
    nt_plant(race)
    target_link_libraries(nt_plant_race PRIVATE Threads::Threads)
else()
    nt_plant(oob)
endif()
